$scripts = @(
    ".\Find-VM.ps1",
    ".\GetServerADGroups.ps1",
    ".\LastBackup.ps1"
)

$vmNames = Get-content "pingServerList2.txt"
$sNo = 0
Clear-Host

foreach ($vmName in $vmNames) {
    $sNo += 1
    $outputs = ""

    foreach ($script in $scripts) {
        Write-Host "PS C:\Scripts> $script $vmName"
        $outputs += "PS C:\Scripts> $script $vmName" | Out-String        
        $outputs += &$script $vmName | Out-String
        Clear-Host
    }

    Set-Clipboard -Value $outputs
    Write-Host "`n`n$sNo `t$vmName copied."

    $Host.UI.ReadLine()
}
