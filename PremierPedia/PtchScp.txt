$vCenters = @(
    "c2pvcenter1.corp.premierinc.com",
    "c3pivcenter1.corp.premierinc.com",
    "clt-cf-vc-01.premierinc.com"
)

foreach ($vCenter in $vCenters) {
    Connect-VIServer $vCenter
    if ($vCenter -eq "c2pvcenter1.corp.premierinc.com") {
        $templates =@(
            "CUL-WIN2019-TAB",
            "CUL-2016-COSTDB",
            "CUL-WIN2022",
            "CUL-WIN2022-ERP",
            "CUL2022-TD530PA",
            "CUL_2019_Template",
            "CUL_2019_CPI_Template",
            "CUL-2019-MEMBER",
            "CUL-2019-SQLDBA",
            "CUL_2016_Tdoc480ProdApp_Template",
            "CUL_2019_Tdoc510ProdApp_Template",
            "CUL2022-TD543PA",
            "CUL2022-TD543STG"            
        )
    } elseif ($vCenter -eq "c3pivcenter1.corp.premierinc.com") {
        $templates =@(
            "CLT22-TD530SAIO",
            "CLT-WIN2022",
            "CLT-WIN22-DMZ",
            "CLT-2019-COSTDB",
            "CLT-2016-COSTDB",
            "CLT-2019-MEMBER",
            "CLT-WIN2019-TAB",
            "CLT_2019_Template",
            "CLT-WIN2022-TAB",
            "CLT_2016_Template"
        )
    } elseif ($vCenter -eq "clt-cf-vc-01.premierinc.com") {
        $templates =@(
            "2019_Template_Old",
            "2019_Tdoc_Template",
            "2019_Template",
            "CF_2022_Template"           
        )
    } else {
        Write-Host "$vCenter is an invalid VC Name"
    }

    foreach ($template in $templates) {
        Set-Template -Template $template -ToVM -Confirm:$false -RunAsync
        Start-sleep -s 15

        Start-VM -VM $template | Get-VMQuestion | Set-VMQuestion -DefaultOption -Confirm:$false
        Start-sleep -s 45

        read-host -assecurestring | convertfrom-securestring | out-file D:\OSPwd.txt
        $Username = "samadmin"
        $OSPwd = cat D:\OSPwd.txt | convertto-securestring
        $cred = new-object -typename System.Management.Automation.PSCredential -argumentlist $username, $OSPwd

        Invoke-VMScript -ScriptType PowerShell -ScriptText "Get-WUInstall –WindowsUpdate –AcceptAll –AutoReboot -install" -VM $template -GuestCredential $Cred | Out-file -Filepath D:\Feb8.log -Append
        Start-sleep -s 120

        Shutdown-VMGuest –VM $template -Confirm:$false
        Start-sleep -s 120
        Set-VM –VM $template -ToTemplate -Confirm:$false
        Write-Host "$template Patched"
    }    
}
